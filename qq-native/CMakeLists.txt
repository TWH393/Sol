cmake_minimum_required(VERSION 3.20)
project(SolQqNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(../vendor/minhook minhook)
add_library(sol-qq-loader SHARED loader.cpp)
target_link_libraries(sol-qq-loader PRIVATE minhook)
target_compile_definitions(sol-qq-loader PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(sol-qq-launcher launcher.cpp)
target_compile_definitions(sol-qq-launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(sol-qq-loader PRIVATE /utf-8)
  target_compile_options(sol-qq-launcher PRIVATE /utf-8)
  set_property(TARGET sol-qq-loader sol-qq-launcher minhook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
